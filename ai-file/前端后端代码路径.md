### `formatDateTime()` 函数逻辑

```typescript
function formatDateTime(raw?: string): string {
  if (!raw) return '-'
  let dateStr = String(raw).trim()
  dateStr = dateStr.replace('T', ' ')      // 去掉 T
  dateStr = dateStr.replace('Z', '')         // 去掉 Z
  if (dateStr.includes('.')) {
    dateStr = dateStr.split('.')[0]  // 去掉毫秒部分
  }
  return dateStr
}
```
